  <script>
    // ÇOKLU SEÇİM KUTUCUKLARI (CHECKBOX) İÇİN YARDIMCI FONKSİYONLAR
    function getCheckboxValues(cbName, digerId, digerTextId) {
      let vals = [];
      document.querySelectorAll(`input[name="${cbName}"]:checked`).forEach(cb => {
        if(cb.id === digerId) {
          let txt = document.getElementById(digerTextId).value.trim();
          if(txt) vals.push(txt);
        } else {
          vals.push(cb.value);
        }
      });
      return vals.length > 0 ? vals.join(", ") : "Yok";
    }

    function setCheckboxValues(cbName, digerId, digerTextId, valueString) {
      document.querySelectorAll(`input[name="${cbName}"]`).forEach(cb => cb.checked = false);
      document.getElementById(digerTextId).style.display = 'none';
      document.getElementById(digerTextId).value = "";
      
      if (!valueString || valueString === "Yok") return;
      
      let vals = valueString.split(",").map(v => v.trim());
      let digerVals = [];
      
      vals.forEach(v => {
        let found = false;
        document.querySelectorAll(`input[name="${cbName}"]`).forEach(cb => {
          if (cb.value === v && cb.id !== digerId) { cb.checked = true; found = true; }
        });
        if (!found && v !== "") digerVals.push(v);
      });
      
      if(digerVals.length > 0) {
        document.getElementById(digerId).checked = true;
        document.getElementById(digerTextId).style.display = 'block';

    function parseKurulDosyalari(s) {
        let tutanaklar = (s['Tutanak Özeti'] || "").split('|||').filter(p => p.trim() !== "");
        let dosyalar = [];
        let normalBuffer = [];

        tutanaklar.forEach(t => {
            if (t.includes('[DİREKT KURUL SEVK]')) {
                dosyalar.push({ tip: 'Direkt Kurul Sevki', tutanaklar: [t] });
            } else {
                normalBuffer.push(t);
                if (normalBuffer.length === 3) {
                    dosyalar.push({ tip: '3 Tutanak Birikimi (Otomatik)', tutanaklar: [...normalBuffer] });
                    normalBuffer = [];
                }
            }
        });
        return dosyalar;
    }

    function applyGlobalFilters(dataList) {
     let kademe = seciliKademe === 'Tümü' ? '' : seciliKademe;
      if (kademe === 'bekle') kademe = ''; 
      const sube = document.getElementById('filterSube').value;
      const arama = document.getElementById('searchInput').value.toLowerCase();
      return dataList.filter(s => {
        const sinif = s['Sınıfı / Şubesi'] || "";
        const c1 = kademe === "" || sinif.startsWith(kademe + "/");
        const c2 = sube === "" || sinif.endsWith("/" + sube) || sinif.endsWith("-" + sube) || sinif.includes(sube);
        const c3 = arama === "" || s['Adı Soyadı'].toLowerCase().includes(arama) || s['Öğrenci No'].toString().includes(arama);
        return c1 && c2 && c3;
      });
    }

    function filterDatabase() {
      const kademeVal = seciliKademe;
      const aramaVal = document.getElementById('searchInput').value.trim();
      
      if(currentModule === 'tutanak') { loadTutanakList(); return; }
      
      const grid = document.getElementById('resultsGrid'); 
      const fixedBar = document.getElementById('fixedActionBar');
      const moduleTabs = document.getElementById('moduleTabs');
      const printActions = document.getElementById('printActions');
      
      if (kademeVal === "bekle" && aramaVal === "") {
        grid.innerHTML = `<div style="grid-column:1/-1; text-align:center; padding:60px 20px; color:var(--text-muted); background:white; border-radius:20px; border:1px dashed #cbd5e1;">

                            <p style="font-size:15px; margin:0;">Öğrencileri listelemek için yukarıdan bir kademe seçin veya isimle arama yapın.</p>
                          </div>`;
        fixedBar.style.display = 'none';
        document.getElementById('printArea').innerHTML = '';
        return;
      }
      
      grid.innerHTML = '';
      fixedBar.style.display = 'flex'; // Kademeyi seçtiğin an butonlar ve filtreler sabit olarak açılır
      
      let filtered = applyGlobalFilters(activeDatabase);
      let tabOptions = [];

      if (currentModule === 'okuma') {
        tabOptions = ['Tümü', 'Bilmiyor', 'Destek Gerekiyor', 'Normal'];
        if (activeSubTab !== 'Tümü') filtered = filtered.filter(s => s['Okuma-Yazma Durumu'] === activeSubTab);
      } 
      else if (currentModule === 'kurslar') {
        tabOptions = ['Tümü', 'DYK', 'Destek Eğitimi', 'METİP Telafi'];
        // 'Yok' olanları ve boş olanları baştan eliyoruz ki sadece kurs/eğitim alanlar listelensin
        filtered = filtered.filter(s => s['Kurs Durumu'] && s['Kurs Durumu'] !== 'Yok' && s['Kurs Durumu'].trim() !== '');
        if (activeSubTab !== 'Tümü') {
          filtered = filtered.filter(s => s['Kurs Durumu'] === activeSubTab);
        }
      }
      else if (currentModule === 'kocluk') {
        filtered = filtered.filter(s => s['Koçluk Öğrencisi'] === 'Evet');
      }
      else if (currentModule === 'risk') {
        tabOptions = ['Tümü', '1. Düzey', '2. Düzey', '3. Düzey'];
        filtered = filtered.filter(s => s['Risk Derecesi'] && s['Risk Derecesi'] !== 'Düşük');
        if (activeSubTab !== 'Tümü') filtered = filtered.filter(s => s['Risk Derecesi'] === activeSubTab);
      }
      else if (currentModule === 'academic') { filtered = filtered.filter(s => s['Akademik Potansiyel Rozeti'] === 'Evet'); }
      else if (currentModule === 'metip') { filtered = filtered.filter(s => s['METİP (Mevsimlik İşçi)'] === 'Evet'); }
      else if (currentModule === 'veli') { 
        tabOptions = ['Tümü', 'Destekleyici / İlgili', 'Sadece Çağrılınca Gelir', 'İlgisiz', 'Savunmacı / Çatışmacı'];
        if (activeSubTab !== 'Tümü') {
          filtered = filtered.filter(s => s['Veli Katılım'] === activeSubTab);
        }
      }
      else if (currentModule === 'devamsiz') { filtered = filtered.filter(s => s['Sürekli Devamsızlık'] === 'Evet'); }
      else if (currentModule === 'kurul') { 
        tabOptions = ['Tümü', 'Karar Bekleyenler', 'Karara Bağlananlar'];
        filtered = filtered.filter(s => {
          let hesap = getKurulHesap(s);
          if(activeSubTab === 'Tümü') return hesap.dosyaHakki > 0 || hesap.kararSayisi > 0;
          if(activeSubTab === 'Karar Bekleyenler') return hesap.beklemede;
          if(activeSubTab === 'Karara Bağlananlar') return !hesap.beklemede && hesap.kararSayisi > 0;
        });
      }

      // Sadece sayfanın kendine özel sekmelerini (Tabs) içine atıyoruz
      let tabsHTML = "";
      if (tabOptions.length > 0) {
        tabOptions.forEach(t => {
          const activeCls = (t === activeSubTab) ? 'active' : '';
          tabsHTML += `<button class="tab-btn ${activeCls}" onclick="setSubTab('${t}')">${t}</button>`;
        });
      }
      moduleTabs.innerHTML = tabsHTML;

      // Tüm listeleme ekranlarında yazdırma butonlarını göster
      printActions.style.display = 'flex';

      if(filtered.length === 0) { 
        grid.innerHTML = '<p style="grid-column:1/-1; text-align:center; color:var(--text-muted); margin-top:40px; font-weight:600; font-size:16px;">Seçimlerinize uygun kayıt bulunamadı.</p>'; 
      } else {
        filtered.forEach(s => grid.innerHTML += buildCard(s));
      }
      buildPrintTable(filtered, currentModule, activeSubTab);
    }

    function saveTutanak() {
      const seciliKutular = document.querySelectorAll('input[name="t_secili_ogrenciler"]:checked');
      if(seciliKutular.length === 0) { showToast("Lütfen listeden en az bir öğrenci seçin!", true); return; }
      
      const tarih = document.getElementById('t_tarih').value; 
      const ogretmen = document.getElementById('t_ogretmen').value;
      const olayRisk = document.getElementById('t_olayRisk').value; 
      const ozet = document.getElementById('t_ozet').value;
      const direktSevk = document.getElementById('t_direkt_kurul').checked;

      if(!tarih || !ozet || !ogretmen) { showToast("Lütfen tarih, öğretmen ve özet kısımlarını boş bırakmayın!", true); return; }
      
      // 1. Seçilen öğrencilerin No ve İsim bilgilerini alıyoruz
      let seciliOgrenciler = [];
      seciliKutular.forEach(kutu => {
        const ogrNo = kutu.value;
        const stData = activeDatabase.find(x => x['Öğrenci No'].toString() === ogrNo.toString());
        const adSoyad = stData ? stData['Adı Soyadı'] : ogrNo;
        seciliOgrenciler.push({ no: ogrNo, ad: adSoyad });
      });

      const btn = document.getElementById('btnTutanak'); btn.innerText = "Toplu İşleniyor..."; btn.disabled = true;
      document.getElementById('loading').style.display = 'block';
      
      let islenenSayi = 0;
      const toplamSayi = seciliOgrenciler.length;
      
      // 2. Her öğrenciye özel metin oluşturup kaydediyoruz
      seciliOgrenciler.forEach(ogr => {
        let ogrenciyeOzelOzet = ozet;
        
        // Eğer grupta 1'den fazla kişi varsa, diğerlerinin adını öğrencinin notuna ekliyoruz
        if (toplamSayi > 1) {
          const digerleri = seciliOgrenciler.filter(x => x.no !== ogr.no).map(x => x.ad).join(", ");
          ogrenciyeOzelOzet += `\n\n👥 [Toplu Olay] Olaya Karışan Diğer Öğrenciler: ${digerleri}`;
        }

        // Direkt kurul sevki varsa en başa ekle
        if (direktSevk) { 
          ogrenciyeOzelOzet = "[DİREKT KURUL SEVK]\n" + ogrenciyeOzelOzet; 
        }

        google.script.run.withSuccessHandler((response) => { 
          islenenSayi++;
          if (islenenSayi === toplamSayi) bitirTopluTutanak(direktSevk, toplamSayi);
        }).withFailureHandler((err) => {
          islenenSayi++; 
          if (islenenSayi === toplamSayi) bitirTopluTutanak(direktSevk, toplamSayi);
        }).addTutanak(ogr.no, tarih, ogretmen, olayRisk, ogrenciyeOzelOzet);
      });
    }

    function exportToExcel() {
      const table = document.querySelector('.print-table');
      if(!table || table.innerHTML.trim() === "") { 
        showToast("İndirilecek veri bulunamadı!", true); 
        return; 
      }
      let html = "<html xmlns:x=\"urn:schemas-microsoft-com:office:excel\">";
      html += "<head><meta charset='utf-8'></head><body><table border='1'>";
      html += table.innerHTML;
      html += "</table></body></html>";
      const blob = new Blob([html], { type: 'application/vnd.ms-excel;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'REUTS_Ogrenci_Listesi.xls';
      a.click();
      URL.revokeObjectURL(url);
    }

  document.querySelectorAll('.edit-karar-secim').forEach(cb => cb.checked = false);

  turListesi.forEach(t => {
      document.querySelectorAll('.edit-karar-secim').forEach(cb => {
          if (cb.value === t) cb.checked = true;
      });
  });

  const regex = /\[(?:Yönlendirilen|Eklenen) Kurumlar:\s*(.*?)\]/g;

      kurumlarStr = match[1];

    function getKurulHesap(s) {
      let tSayisi = 0; let direktSayisi = 0;
      if (s['Tutanak Özeti']) {
          let parcalar = s['Tutanak Özeti'].split('|||').filter(p => p.trim() !== "");
          tSayisi = parcalar.length;
          direktSayisi = parcalar.filter(p => p.includes('[DİREKT KURUL SEVK]')).length;
      }
      
      // Düzeltilen kısım: Normal tutanak sayısını ayırıyoruz
      let normalTutanakSayisi = tSayisi - direktSayisi; 
      let dosyaHakki = Math.floor(normalTutanakSayisi / 3) + direktSayisi;
      
      let kVerisi = s['Kurul Kararları'] || "";
      let kararSayisi = kVerisi.split('|||').filter(p => p.trim() !== "").length;
      return { dosyaHakki, kararSayisi, beklemede: dosyaHakki > kararSayisi };
    }

    // Form açıkken anlık net ve puan hesaplama motoru (Kademe Katsayılarına Göre Kusursuz Matematik)
    document.addEventListener('input', function(e) {
      if(e.target.classList.contains('k_input')) {
        let topNet = 0;
        let pT = 0, pM = 0, pF = 0, pS = 0, pD = 0, pI = 0;

        ['tr', 'mat', 'fen', 'sos', 'din', 'ing'].forEach(kod => {
          let maxSoru = parseInt(document.getElementById(`k_d_${kod}`).max) || 20;
          let d = parseInt(document.getElementById(`k_d_${kod}`).value) || 0;
          let y = parseInt(document.getElementById(`k_y_${kod}`).value) || 0;
          
          // Hata Önleyici: Girilen değer maksimum soru sayısını geçemez
          if(d > maxSoru) { d = maxSoru; document.getElementById(`k_d_${kod}`).value = maxSoru; }
          if(d + y > maxSoru) { y = maxSoru - d; document.getElementById(`k_y_${kod}`).value = y; }

          let net = d - (y / 3);
          topNet += net;
          
          if(kod==='tr') pT = net; else if(kod==='mat') pM = net; else if(kod==='fen') pF = net;
          else if(kod==='sos') pS = net; else if(kod==='din') pD = net; else if(kod==='ing') pI = net;
        });

        let isKucukKademe = (parseInt(document.getElementById('k_d_tr').max) === 15);
        let tahminiPuan = 195; // MEB Taban Puanı

        if (isKucukKademe) {
          // 5 ve 6. Sınıflar (75 Soru) - Kusursuz Katsayı: Ana Ders (17/3), Ara Ders (5/3)
          // 45 Ana Net + 30 Ara Net = Tam 305 Puan + 195 Taban = 500 Puan
          tahminiPuan += ((pT + pM + pF) * (17/3)) + ((pS + pD + pI) * (5/3));
        } else {
          // 7 ve 8. Sınıflar (90 Soru) - Kusursuz Katsayı: Ana Ders (13/3), Ara Ders (1.5)
          // 60 Ana Net + 30 Ara Net = Tam 305 Puan + 195 Taban = 500 Puan
          tahminiPuan += ((pT + pM + pF) * (13/3)) + ((pS + pD + pI) * 1.5);
        }

        // Güvenlik sınırları
        if(tahminiPuan > 500) tahminiPuan = 500;
        if(tahminiPuan < 195) tahminiPuan = 195;

        document.getElementById('k_canli_net').innerText = topNet.toFixed(2);
        document.getElementById('k_canli_puan').innerText = tahminiPuan.toFixed(2);
      }
    });

    function renderAILgsKarnesi(guncelDeneme, oncekiDeneme) {
      let tarihStr = guncelDeneme['Eklenme Tarihi'] || "";
      if(tarihStr.includes('GMT') || tarihStr.includes('T') || tarihStr.length > 15) {
        try { 
          let dObj = new Date(tarihStr);
          if(!isNaN(dObj.getTime())) {
            tarihStr = dObj.toLocaleDateString('tr-TR');
          }
        } catch(e){}
      }

      const calcNet = (dyStr) => { 
        if(!dyStr && dyStr !== 0) return "0.00";
        let str = dyStr.toString();
        if(str.includes('-')) {
          let [d, y] = str.split('-');
          return (parseInt(d || 0) - (parseInt(y || 0)/3)).toFixed(2);
        }
        return parseFloat(str || 0).toFixed(2);
      };

      let toplamSkor = guncelDeneme['Toplam Net'] || 0;
      let oncekiSkor = oncekiDeneme ? (oncekiDeneme['Toplam Net'] || 0) : null;
      let genelFarkHtml = "";
      
      if(oncekiSkor) {
        let fark = (toplamSkor - oncekiSkor).toFixed(2);
        if(fark > 0) genelFarkHtml = `<span style="color:#10b981;">(Önceki Denemeye Göre: +${fark} Net 📈)</span>`;
        else if(fark < 0) genelFarkHtml = `<span style="color:#ef4444;">(Önceki Denemeye Göre: ${fark} Net 📉)</span>`;
        else genelFarkHtml = `<span style="color:#64748b;">(Önceki Denemeye Göre: Aynı ➖)</span>`;
      }

      const dersler = [
        { ad: '🇹🇷 Türkçe', val: guncelDeneme['Türkçe'], eskiVal: oncekiDeneme ? oncekiDeneme['Türkçe'] : null },
        { ad: '🏛️ İnkılap', val: guncelDeneme['Sosyal'], eskiVal: oncekiDeneme ? oncekiDeneme['Sosyal'] : null },
        { ad: '🕌 Din', val: guncelDeneme['Din Kültürü'], eskiVal: oncekiDeneme ? oncekiDeneme['Din Kültürü'] : null },
        { ad: '🌍 İng', val: guncelDeneme['İngilizce'], eskiVal: oncekiDeneme ? oncekiDeneme['İngilizce'] : null },
        { ad: '📐 Mat', val: guncelDeneme['Matematik'], eskiVal: oncekiDeneme ? oncekiDeneme['Matematik'] : null },
        { ad: '🔬 Fen', val: guncelDeneme['Fen Bilimleri'], eskiVal: oncekiDeneme ? oncekiDeneme['Fen Bilimleri'] : null }
      ];

      let dersHtml = "";
      let analizDersleri = [];

      dersler.forEach(d => {
        let net = parseFloat(calcNet(d.val));
        let eskiNet = d.eskiVal !== null ? parseFloat(calcNet(d.eskiVal)) : net;

        if(oncekiDeneme) analizDersleri.push({ ad: d.ad.substring(3), fark: parseFloat(fark) });

    // Tarihi sıralayabilmek için gerçek formata çeviren yardımcı motor
    function parseTarihToDate(tarihStr) {
        if(!tarihStr) return new Date(0);
        if(tarihStr.includes('.')) {
            let p = tarihStr.split('.');
            if(p.length === 3) return new Date(`${p[2]}-${p[1]}-${p[0]}T00:00:00`);
        }
        let parsed = new Date(tarihStr);
        return isNaN(parsed.getTime()) ? new Date(0) : parsed;
    }

            if(!val && val !== 0) return 0;

        });

    // 3. İŞTE RESİMDEKİ GİBİ SIFIRDAN KUSURSUZ TABLOYU BASAN MOTOR
    function extractDY(val) {
        if(!val && val !== 0) return {d:0, y:0, n:0};
        let str = val.toString();
        if(str.includes('-')) {
            let p = str.split('-');
            let d = parseInt(p[0]||0); let y = parseInt(p[1]||0);
            return { d: d, y: y, n: (d - (y/3)) };
        }
        return { d: 0, y: 0, n: parseFloat(str||0) };
    }

// DÜZENLEME MOTORU
function editDeneme(ogrNo, tarih, ad) {
    let deneme = globalKoclukDenemeler.find(d => d['Öğrenci No'].toString() === ogrNo.toString() && (d['Sınav Tarihi'] === tarih || d['Eklenme Tarihi'] === tarih) && d['Deneme Adı'] === ad);
    if(!deneme) return;

    document.getElementById('edit_kocluk_modu').value = "aktif";
    activeEditEskiTarih = tarih;
    activeEditEskiAd = ad;

    document.getElementById('secili_ogr_no').value = ogrNo;
    
    // Tarihi HTML input'un anlayacağı YYYY-MM-DD formatına çeviren motor
    let formatliTarih = tarih;
    if (tarih) {
        let dObj = new Date(tarih);
        if (!isNaN(dObj.getTime())) { // Eğer geçerli bir tarih ise çevir
            let y = dObj.getFullYear();
            let m = ("0" + (dObj.getMonth() + 1)).slice(-2);
            let d = ("0" + dObj.getDate()).slice(-2);
            formatliTarih = `${y}-${m}-${d}`;
        }
    }
    
    document.getElementById('k_tarih').value = formatliTarih;
    document.getElementById('k_ad').value = ad;

    let s = activeDatabase.find(x => x['Öğrenci No'].toString() === ogrNo.toString());
    let sinif = s ? s['Sınıfı / Şubesi'] : "8";
    let isKucukKademe = sinif.startsWith("5") || sinif.startsWith("6");
    let anaDersMax = isKucukKademe ? 15 : 20;

    ['tr', 'mat', 'fen'].forEach(kod => {
        let max = anaDersMax;
        if(document.getElementById(`k_d_${kod}`)) document.getElementById(`k_d_${kod}`).max = max;
        if(document.getElementById(`k_y_${kod}`)) document.getElementById(`k_y_${kod}`).max = max;
    });

    const kodMap = {'tr':'Türkçe', 'mat':'Matematik', 'fen':'Fen Bilimleri', 'sos':'Sosyal', 'din':'Din Kültürü', 'ing':'İngilizce'};
    
    Object.keys(kodMap).forEach(kod => {
        let valStr = (deneme[kodMap[kod]] || "").toString();
        let d = "", y = "";
        if(valStr.includes('-')) { d = valStr.split('-')[0]; y = valStr.split('-')[1]; } 
        else { d = valStr; }
        if(document.getElementById(`k_d_${kod}`)) document.getElementById(`k_d_${kod}`).value = d;
        if(document.getElementById(`k_y_${kod}`)) document.getElementById(`k_y_${kod}`).value = y;
    });

    document.getElementById('k_d_tr').dispatchEvent(new Event('input', { bubbles: true }));
    document.getElementById('denemeModal').style.display = 'flex';
}

function gonderPdrSevk() {
  const ogrenciGirdisi = document.getElementById('pdr_sevk_ogrenci').value.trim();
  const ogretmen = document.getElementById('pdr_sevk_ogretmen').value.trim();
  const rehberOgr = document.getElementById('pdr_sevk_hedef').value;
  const konu = document.getElementById('pdr_sevk_konu').value.trim();
  const riskDuzeyi = document.getElementById('pdr_sevk_risk') ? document.getElementById('pdr_sevk_risk').value : "Düşük"; 

  // KATEGORİYİ (KONU BAŞLIĞI) YAKALAMA
  let kategori = "Genel Dosya";
  if (document.getElementById('pdr_sevk_kategori')) {
    kategori = document.getElementById('pdr_sevk_kategori').value;
    if (kategori === 'Diğer') {
      kategori = document.getElementById('pdr_sevk_kategori_diger').value.trim() || "Diğer Konular";
    }
  }

  if(!ogrenciGirdisi || !ogrenciGirdisi.includes(' - ') || !ogretmen || !rehberOgr || !konu) {
    showToast("Lütfen tüm alanları doldurun ve öğrenciyi listeden tam seçin!", true); return;
  }

  const parcalar = ogrenciGirdisi.split(' - ');
  const ogrNo = parcalar[0].trim();
  const ogrAd = parcalar[1].split(" (")[0].trim();

  const btn = document.getElementById('btnPdrSevk');
  btn.innerHTML = "<i class='fas fa-spinner fa-spin'></i> Kayıtlar Taranıyor...";
  btn.disabled = true;

  // ARKA PLANDAKİ ayniSevkVarMi MOTORUNU GERÇEKTEN ÇAĞIRIYORUZ
  google.script.run
    .withSuccessHandler(function(ayniDosyaVarMi) {
       if (ayniDosyaVarMi) {
         // Aynı kategoride aktif/devam eden dosya varsa işlemi durdur ve uyarıyı çıkar
         btn.innerHTML = "<i class='fas fa-paper-plane'></i> Yönlendirmeyi PDR Servisine Gönder";
         btn.disabled = false;
         
         showConfirm(
           `Bu öğrenci için "${kategori}" konu başlığında zaten PDR'de aktif veya devam eden bir sevk dosyası bulunuyor. Yine de mükerrer olarak bu sevki göndermek istiyor musunuz?`, 
           function() {
             sevkIsleminiTamamla(ogrNo, ogrAd, ogretmen, rehberOgr, kategori, konu, riskDuzeyi);
           }, 
           "Yine de Gönder", 
           "fa-paper-plane"
         );
       } else {
         // Aynı kategoride açık dosya yoksa direkt arka planda kaydet
         sevkIsleminiTamamla(ogrNo, ogrAd, ogretmen, rehberOgr, kategori, konu, riskDuzeyi);
       }
    })

       btn.innerHTML = "<i class='fas fa-paper-plane'></i> Yönlendirmeyi PDR Servisine Gönder";
       btn.disabled = false;
       showToast("Sistem kontrolünde hata oluştu!", true);
    })
    .ayniSevkVarMi(ogrNo, kategori);
}

function formatMetipDateForInput(dateStr) {
    if (!dateStr) return "";
    try {
        if (typeof dateStr === 'string' && dateStr.match(/^\d{4}-\d{2}-\d{2}$/)) {
            let p = dateStr.split('-');
            return `${p[2]}.${p[1]}.${p[0]}`;
        }
        let d = new Date(dateStr);
        if (!isNaN(d.getTime())) {
            let y = d.getFullYear();
            let m = ("0" + (d.getMonth() + 1)).slice(-2);
            let day = ("0" + d.getDate()).slice(-2);
            return `${day}.${m}.${y}`;
        }
    } catch(e) {}
    return "";
}

// Sekmeye göre dinamik çalışan Excel (CSV) İndirme Fonksiyonu
function exportMetipExcel() {
    let liste = getFilteredMetipOgrencileri();
    if(liste.length === 0) {
        alert("İndirilecek öğrenci bulunamadı.");
        return;
    }

    let csvContent = "\uFEFF"; 
    
    if (activeMetipSekme === 'Detaylı Liste') {
        csvContent += "Ogrenci No;Adi Soyadi;Sinifi;Durumu;Geldigi Tarih;Gittigi Tarih;Gittigi Yer\n";
        liste.forEach(s => {
            let row = [
                s['Öğrenci No'] || "",
                `"${s['Adı Soyadı'] || ""}"`,
                `"${s['Sınıfı / Şubesi'] || ""}"`,
                `"${s['METİP Durumu'] || ""}"`,
                `"${formatMetipDateForInput(s['METİP Geldiği Tarih']) || ""}"`,
                `"${formatMetipDateForInput(s['METİP Gittiği Tarih']) || ""}"`,
                `"${s['METİP Gittiği Yer'] || ""}"`
            ].join(";");
            csvContent += row + "\n";
        });
    } else {
        let basliklar = ["Sira", "Ogrenci No", "Adi Soyadi", "Sinifi"];
        if (activeMetipSekme === 'Saf Liste') basliklar.push("Durumu");
        csvContent += basliklar.join(";") + "\n";
        
        liste.forEach((s, index) => {
            let row = [
                index + 1,
                s['Öğrenci No'] || "",
                `"${s['Adı Soyadı'] || ""}"`,
                `"${s['Sınıfı / Şubesi'] || ""}"`
            ];
            if (activeMetipSekme === 'Saf Liste') row.push(`"${s['METİP Durumu'] || ""}"`);
            csvContent += row.join(";") + "\n";
        });
    }

    let blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    let url = URL.createObjectURL(blob);
    let link = document.createElement("a");
    link.setAttribute("href", url);
    let dosyaAdi = `METIP_Listesi_${activeMetipSekme.replace(/ /g, "_")}.csv`;
    link.setAttribute("download", dosyaAdi);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
}

    let kilitZamani = parseInt(localStorage.getItem("reuts_lock_time")) || 0;
    let simdi = new Date().getTime();
    const btn = document.getElementById("btnLogin");
    const errorBox = document.getElementById("loginError");

    if (hataSayisi >= 3) {
        if (simdi - kilitZamani < 900000) {
            let kalanDakika = Math.ceil((900000 - (simdi - kilitZamani)) / 60000);
            btn.disabled = true; btn.style.background = "#94a3b8";
            errorBox.style.display = "block";
            errorBox.innerText = `Çok fazla hatalı deneme! Lütfen ${kalanDakika} dakika bekleyin.`;
            return true;
        } else {
            localStorage.setItem("reuts_fail_count", 0);
            localStorage.setItem("reuts_lock_time", 0);
            btn.disabled = false; btn.style.background = "linear-gradient(135deg, #3b82f6, #1d4ed8)";
            errorBox.style.display = "none";
            return false;
        }
    }
    return false;
}

function otomatikIsimDoldur() {
    const aktifOgretmen = localStorage.getItem("reuts_active_teacher") || "";
    
    // Tutanak Modülündeki Öğretmen Kutusu
    const tutanakOgr = document.getElementById("t_ogretmen");
    if (tutanakOgr && aktifOgretmen) {
        tutanakOgr.value = aktifOgretmen;
        tutanakOgr.readOnly = true; // Değiştirmeyi engelle
        tutanakOgr.style.background = "#e2e8f0"; // Kapalı olduğunu belli etmek için gri yap
        tutanakOgr.style.cursor = "not-allowed";
    }

    // PDR Sevk Modülündeki Öğretmen Kutusu
    const pdrOgr = document.getElementById("pdr_sevk_ogretmen");
    if (pdrOgr && aktifOgretmen) {
        pdrOgr.value = aktifOgretmen;
        pdrOgr.readOnly = true; 
        pdrOgr.style.background = "#e2e8f0";
        pdrOgr.style.cursor = "not-allowed";
    }
}

function sistemeGirisYap() {
    if(kilitKontroluYap()) return; 

    const user = document.getElementById("loginUser").value.trim();
    const pass = document.getElementById("loginPass").value.trim();
    const errorBox = document.getElementById("loginError");
    const btn = document.getElementById("btnLogin");

    if(!user || !pass) {
        errorBox.style.display = "block"; errorBox.innerText = "Kullanıcı adı ve şifre boş bırakılamaz!";
        return;
    }

    btn.innerHTML = "<i class='fas fa-spinner fa-spin'></i> Kontrol Ediliyor..."; 
    btn.disabled = true;
    errorBox.style.display = "none";

    google.script.run
    .withSuccessHandler(function(response) {
        btn.innerHTML = "<i class='fas fa-sign-in-alt'></i> Giriş Yap"; 
        btn.disabled = false;

        if(response.success) {
            localStorage.setItem("reuts_fail_count", 0);
            localStorage.setItem("reuts_active_teacher", response.name); 
            localStorage.setItem("reuts_role", response.role); 
            
            document.getElementById("loginOverlay").style.display = "none"; 
            otomatikIsimDoldur();
            setTimeout(function() {
                window.scrollTo(0, 0);
                document.body.scrollTop = 0;
                document.documentElement.scrollTop = 0;
            }, 100);
        } else {
            let hataSayisi = parseInt(localStorage.getItem("reuts_fail_count")) || 0;
            hataSayisi++; localStorage.setItem("reuts_fail_count", hataSayisi);
            
            errorBox.style.display = "block";

            if(hataSayisi >= 3) {
                localStorage.setItem("reuts_lock_time", new Date().getTime());
                kilitKontroluYap();
            } else {
                errorBox.innerText = `Hatalı şifre! (Kalan deneme hakkı: ${3 - hataSayisi})`;
            }
        }
    })
    .withFailureHandler(function(err) {
        btn.innerHTML = "<i class='fas fa-sign-in-alt'></i> Giriş Yap"; 
        btn.disabled = false;
        errorBox.style.display = "block";
        errorBox.innerText = "Bağlantı hatası oluştu. Lütfen tekrar deneyin.";
    })
    .checkLoginCredentials(user, pass);
}

function cikisYap() {
    localStorage.removeItem("reuts_active_teacher");
    localStorage.removeItem("reuts_role");

    document.getElementById("loginUser").value = "";
    document.getElementById("loginPass").value = "";
    
    const btn = document.getElementById("btnLogin");
    if(btn) {
        btn.innerHTML = "<i class='fas fa-sign-in-alt'></i> Giriş Yap";
        btn.disabled = false;
    }

    document.getElementById("loginOverlay").style.display = "flex";
    loadModule('dashboard'); 
}
